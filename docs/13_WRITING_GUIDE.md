# 13 - WRITING GUIDE

Version: 1.0

---

# هدف

این سند استاندارد نگارش تمام مقالات CafeAmoozesh را مشخص می‌کند.

هدف فقط تولید متن نیست.

هدف تولید محتوایی است که:

- روان باشد.
- حرفه‌ای باشد.
- برای کاربر قابل فهم باشد.
- برای گوگل قابل درک باشد.

---

# اصل طلایی

همیشه برای انسان بنویس.

سپس برای موتور جستجو بهینه کن.

هرگز برعکس عمل نکن.

---

# لحن مقاله

لحن تمام مقالات باید:

- دوستانه
- حرفه‌ای
- آموزشی
- شفاف
- صمیمی

باشد.

---

# طول جملات

پیشنهاد:

۱۰ تا ۲۰ کلمه

حداکثر:

۳۰ کلمه

اگر جمله طولانی شد، آن را به دو جمله تقسیم کن.

---

# طول پاراگراف

حداکثر:

۵ جمله

پیشنهاد:

۲ تا ۴ جمله

---

# شروع جملات

چهار جمله متوالی نباید با یک کلمه شروع شوند.

نمونه اشتباه:

JavaScript ...

JavaScript ...

JavaScript ...

JavaScript ...

---

# کلمات انتقالی

در متن از کلمات انتقالی استفاده شود.

نمونه:

- بنابراین
- در نتیجه
- از طرف دیگر
- علاوه بر این
- برای مثال
- به همین دلیل
- با این حال
- اکنون
- در ادامه

---

# اصطلاحات انگلیسی

اولین بار:

نام فارسی + نام انگلیسی

نمونه:

پشته فراخوانی (Call Stack)

از دفعات بعد:

Call Stack

یا

پشته فراخوانی

---

# Bold

برای:

- مفاهیم مهم
- اصطلاحات کلیدی
- نتیجه‌گیری

از

<strong>

استفاده شود.

---

# Italic

برای اصطلاحات خارجی

از

<em>

استفاده شود.

---

# لیست‌ها

اگر بیش از دو مورد وجود دارد،

از لیست استفاده شود.

---

# جدول

اگر قرار است مقایسه انجام شود،

از جدول استفاده شود.

---

# مثال

هر مفهوم جدید باید حداقل یک مثال داشته باشد.

---

# مثال واقعی

مثال‌ها باید به پروژه واقعی نزدیک باشند.

از مثال‌های مصنوعی خودداری شود.

---

# کد

قبل از کد:

توضیح

بعد از کد:

تحلیل

---

# تصویر

هر 400 تا 600 کلمه

یک تصویر

---

# CTA

ابتدای مقاله

↓

انتهای مقاله

وجود داشته باشد.

---

# سوال از کاربر

در صورت نیاز برای درگیر کردن مخاطب استفاده شود.

نمونه:

آیا تا به حال با این خطا مواجه شده‌اید؟

---

# جمع‌بندی

هر مقاله باید:

- نکات مهم را مرور کند.
- مسیر یادگیری بعدی را معرفی کند.

---

# غلط‌های رایج

قبل از انتشار بررسی شود:

- نیم‌فاصله
- فاصله قبل و بعد از علائم
- املای صحیح اصطلاحات
- نگارش یکدست

---

# نیم‌فاصله

در تمام متن یکدست استفاده شود.

---

.

---

# استفاده از ایموجی

در متن آموزشی استفاده نشود.

فقط در CTA یا باکس‌های اطلاع‌رسانی در صورت نیاز.

---

# ممنوعیت‌ها

- ترجمه ماشینی
- پاراگراف‌های بلند
- جملات بسیار طولانی
- اصطلاحات بدون توضیح
- تکرار زیاد کلمات
- لحن غیررسمی بیش از حد
